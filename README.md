## 🐹 GO MISSION LOG

```go
package main

import (
	"fmt"
	"sync"
)

type Engineer struct {
	Name   string
	Focus  []string
	Status string
}

func main() {
	me := Engineer{
		Name:   "Chirabrata Ghosal",
		Focus:  []string{"Agentic AI", "RAG", "Scalable Backends"},
		Status: "Open to internships",
	}

	var wg sync.WaitGroup
	for _, skill := range me.Focus {
		wg.Add(1)
		go func(s string) {
			defer wg.Done()
			fmt.Printf("⚡ Leveling up: %s\n", s)
		}(skill)
	}
	wg.Wait()

	fmt.Println("🚀", me.Name, "|", me.Status)
}
```
